practice-2
---------------------

mkdir jenkins
cd jenkins
vi dockerfile

FROM jenkins/jenkins:lts
ENV JAVA_OPTS="-Djenkins.install.runSetupWizard=false"
USER root
RUN apt-get update && \
    apt-get install -y sudo git curl docker.io && \
    rm -rf /var/lib/apt/lists/*
USER jenkins
EXPOSE 8080
CMD ["jenkins.sh"]



FROM jenkins/jenkins:lts
ENV JAVA_OPTS="-Djenkins.install.runSetupWizard=false"
USER root
RUN apt-get update && \
    apt-get install -y \
        openjdk-21-jre \
        docker.io \
        git \
        curl \
        sudo \
    && rm -rf /var/lib/apt/lists/*

ENV JAVA_HOME=/usr/lib/jvm/java-21-openjdk-amd64
ENV PATH=$JAVA_HOME/bin:$PATH
USER jenkins
EXPOSE 8080
CMD ["jenkins.sh"]

docker image build -t jenkins:1.0 .
